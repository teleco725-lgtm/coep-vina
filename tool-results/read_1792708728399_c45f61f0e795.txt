   128→async function fetchFichas(
   129→  filters: Record<string, string | undefined>,
   130→): Promise<FichaFIBER[]> {
   131→  const params = new URLSearchParams();
   132→  for (const [k, v] of Object.entries(filters)) {
   133→    if (v && v !== FILTER_ALL) params.set(k, v);
   134→  }
   135→  params.set('limit', '200');
   136→  const res = await fetch(`/api/fichas?${params.toString()}`, {
   137→    cache: 'no-store',
   138→  });
   139→  if (!res.ok) throw new Error('No se pudieron cargar las fichas');
   140→  return (await res.json()) as FichaFIBER[];
   141→}

   407→// ---------------------------------------------------------------------------
   408→// New ficha dialog (react-hook-form + zod)
   409→// ---------------------------------------------------------------------------
   410→function NewFichaDialog({
   411→  open,
   412→  onOpenChange,
   413→  onCreated,
   414→}: {
   415→  open: boolean;
   416→  onOpenChange: (v: boolean) => void;
   417→  onCreated: () => void;
   418→}) {
   419→  const [submitting, setSubmitting] = useState(false);
   420→  const form = useForm<NewFichaForm>({
   421→    resolver: zodResolver(newFichaSchema),
   422→    defaultValues: {
   423→      fullName: '',
   424→      rut: '',
   425→      address: '',
   426→      commune: 'Viña del Mar',
   427→      lat: null,
   428→      lng: null,
   429→      phone: '',
   430→      email: '',
   431→      ageRange: null,
   432→      riskLevel: 'bajo',
   433→      vulnerability: '',
   434→      householdSize: 1,
   435→      hasPets: false,
   436→      needsMedication: false,
   437→      needsMobility: false,
   438→      needsEvacuation: false,
   439→      notes: '',
   440→      assignedTo: '',
   441→    },
   442→  });
   443→
   444→  const onSubmit = async (values: NewFichaForm) => {
   445→    setSubmitting(true);
   446→    try {
   447→      const body = {
   448→        ...values,
   449→        email: values.email === '' ? null : values.email,
   450→        lat: values.lat === null || Number.isNaN(values.lat as number) ? null : values.lat,
   451→        lng: values.lng === null || Number.isNaN(values.lng as number) ? null : values.lng,
   452→      };
   453→      const res = await fetch('/api/fichas', {
   454→        method: 'POST',
   455→        headers: { 'Content-Type': 'application/json' },
   456→        body: JSON.stringify(body),
   457→      });
   458→      if (!res.ok) {
   459→        const err = (await res.json().catch(() => null)) as {
   460→          error?: string;
   461→          issues?: { message: string }[];
   462→        } | null;
   463→        const msg =
   464→          err?.error ??
   465→          err?.issues?.[0]?.message ??
   466→          'No se pudo crear la ficha.';
   467→        toast.error('Error al crear ficha', { description: msg });
   468→        return;
   469→      }
   470→      const created = (await res.json()) as FichaFIBER;
   471→      toast.success('Ficha creada', {
   472→        description: `Código ${created.code}`,
   473→      });
   474→      form.reset();
   475→      onCreated();
   476→    } catch {
   477→      toast.error('Error de red al crear la ficha.');
   478→    } finally {
   479→      setSubmitting(false);
   480→    }
   481→  };
   482→
   483→  return (
   484→    <Dialog open={open} onOpenChange={onOpenChange}>
   485→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl">
   486→        <DialogHeader>
   487→          <DialogTitle>Nueva Ficha FIBER</DialogTitle>
   488→          <DialogDescription>
   489→            Registro individual de bienestar y riesgo. Datos sensibles bajo Ley
   490→            19.628 — requiere consentimiento informado del titular.
   491→          </DialogDescription>
   492→        </DialogHeader>
   493→
   494→        <Form {...form}>
   495→          <form
   496→            onSubmit={form.handleSubmit(onSubmit)}
   497→            className="grid grid-cols-1 gap-3 sm:grid-cols-2"
   498→          >
   499→            <FormField
   500→              control={form.control}
   501→              name="fullName"
   502→              render={({ field }) => (
   503→                <FormItem className="sm:col-span-2">
   504→                  <FormLabel>Nombre completo *</FormLabel>
   505→                  <FormControl>
   506→                    <Input placeholder="Ej. María González Pérez" {...field} />
   507→                  </FormControl>
   508→                  <FormMessage />
   509→                </FormItem>
   510→              )}
   511→            />
   512→            <FormField
   513→              control={form.control}
   514→              name="rut"
   515→              render={({ field }) => (
   516→                <FormItem>
   517→                  <FormLabel>RUT</FormLabel>
   518→                  <FormControl>
   519→                    <Input placeholder="12.345.678-9" {...field} />
   520→                  </FormControl>
   521→                  <FormMessage />
   522→                </FormItem>
   523→              )}
   524→            />
   525→            <FormField
   526→              control={form.control}
   527→              name="ageRange"
   528→              render={({ field }) => (
   529→                <FormItem>
   530→                  <FormLabel>Rango etario</FormLabel>
   531→                  <Select
   532→                    onValueChange={(v) => field.onChange(v as AgeRange)}
   533→                    value={field.value ?? undefined}
   534→                  >
   535→                    <FormControl>
   536→                      <SelectTrigger className="w-full">
   537→                        <SelectValue placeholder="Selecciona…" />
   538→                      </SelectTrigger>
   539→                    </FormControl>
   540→                    <SelectContent>
   541→                      {AGE_RANGES.map((r) => (
   542→                        <SelectItem key={r} value={r}>
   543→                          {r} años
   544→                        </SelectItem>
   545→                      ))}
   546→                    </SelectContent>

   548→                  <FormMessage />
   549→                </FormItem>

   551→            />
   552→            <FormField
   553→              control={form.control}
   554→              name="address"
   555→              render={({ field }) => (
   556→                <FormItem className="sm:col-span-2">
   557→                  <FormLabel>Dirección</FormLabel>

   562→                    />
   563→                  </FormControl>
   564→                  <FormMessage />
   565→                </FormItem>
   566→              )}
   567→            />
   568→            <FormField
   569→              control={form.control}
   570→              name="commune"
   571→              render={({ field }) => (
   572→                <FormItem>
   573→                  <FormLabel>Comuna</FormLabel>
   574→                  <Select onValueChange={field.onChange} value={field.value}>
   575→                    <FormControl>
   576→                      <SelectTrigger className="w-full">
   577→                        <SelectValue />
   578→                      </SelectTrigger>
   579→                    </FormControl>
   580→                    <SelectContent>
   581→                      {COMMUNES.map((c) => (
   582→                        <SelectItem key={c} value={c}>
   583→                          {c}
   584→                        </SelectItem>
   585→                      ))}
   586→                    </SelectContent>
   587→                  </Select>
   588→                  <FormMessage />
   589→                </FormItem>
   590→              )}
   591→            />
   592→            <FormField
   593→              control={form.control}
   594→              name="riskLevel"
   595→              render={({ field }) => (
   596→                <FormItem>
   597→                  <FormLabel>Nivel de riesgo</FormLabel>
   598→                  <Select
   599→                    onValueChange={field.onChange}
   600→                    value={field.value}
   601→                  >

   815→function EditFichaDialog({
   816→  ficha,
   817→  onClose,
   818→  onUpdated,
   819→}: {
   820→  ficha: FichaFIBER;
   821→  onClose: () => void;
   822→  onUpdated: () => void;
   823→}) {
   824→  const [submitting, setSubmitting] = useState(false);
   825→  const form = useForm<EditFichaForm>({
   826→    resolver: zodResolver(editFichaSchema),
   827→    defaultValues: {
   828→      riskLevel: ficha.riskLevel,
   829→      status: ficha.status,
   830→      assignedTo: ficha.assignedTo ?? '',
   831→      notes: ficha.notes ?? '',
   832→    },
   833→  });
   834→
   835→  // Reset form when ficha changes
   836→  useEffect(() => {
   837→    form.reset({
   838→      riskLevel: ficha.riskLevel,
   839→      status: ficha.status,
   840→      assignedTo: ficha.assignedTo ?? '',
   841→      notes: ficha.notes ?? '',
   842→    });
   843→  }, [ficha, form]);
   844→
   845→  const onSubmit = async (values: EditFichaForm) => {
   846→    setSubmitting(true);
   847→    try {
   848→      const body = {
   849→        riskLevel: values.riskLevel,
   850→        status: values.status,
   851→        assignedTo: values.assignedTo || null,
   852→        notes: values.notes || null,
   853→      };
   854→      const res = await fetch(`/api/fichas/${ficha.id}`, {
   855→        method: 'PATCH',
   856→        headers: { 'Content-Type': 'application/json' },
   857→        body: JSON.stringify(body),
   858→      });
   859→      if (!res.ok) {
   860→        const err = (await res.json().catch(() => null)) as {
   861→          error?: string;
   862→        } | null;
   863→        toast.error('Error al actualizar', {
   864→          description: err?.error ?? 'Intenta de nuevo.',
   865→        });
   866→        return;
   867→      }
   868→      toast.success('Ficha actualizada', {
   869→        description: `${ficha.code} · ${riskLevelLabel(values.riskLevel)} · ${fichaStatusLabel(
   870→          values.status,
   871→        )}`,
   872→      });
   873→      onUpdated();
   874→    } catch {
   875→      toast.error('Error de red al actualizar.');
   876→    } finally {
   877→      setSubmitting(false);
   878→    }
   879→  };
   880→
   881→  return (
   882→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   883→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-xl">
   884→        <DialogHeader>
   885→          <DialogTitle className="flex items-center gap-2">
   886→            <FileText className="h-4 w-4 text-primary" />
   887→            {ficha.code}
   888→          </DialogTitle>
   889→          <DialogDescription>
   890→            {ficha.fullName} · {ficha.commune}
   891→          </DialogDescription>
   892→        </DialogHeader>
   893→
   894→        {/* Read-only detail */}
   895→        <div className="grid grid-cols-1 gap-2 rounded-md border bg-muted/30 p-3 text-xs sm:grid-cols-2">
   896→          {ficha.rut && (
   897→            <DetailRow icon={IdCard} label="RUT" value={ficha.rut} />
   898→          )}
   899→          {ficha.address && (
   900→            <DetailRow icon={MapPin} label="Dirección" value={`${ficha.address}, ${ficha.commune}`} />
   901→          )}
   902→          {ficha.phone && (
   903→            <DetailRow icon={Phone} label="Teléfono" value={ficha.phone} />
   904→          )}
   905→          {ficha.email && (
   906→            <DetailRow icon={Mail} label="Email" value={ficha.email} />
   907→          )}
   908→          {ficha.ageRange && (
   909→            <DetailRow icon={Users} label="Edad" value={`${ficha.ageRange} años`} />
   910→          )}
   911→          <DetailRow
   912→            icon={Users}
   913→            label="Hogar"
   914→            value={`${ficha.householdSize} personas`}
   915→          />
   916→          {ficha.vulnerability && (
   917→            <DetailRow icon={Accessibility} label="Vulnerabilidad" value={ficha.vulnerability} />
   918→          )}
   919→        </div>
   920→
   921→        <Form {...form}>
   922→          <form
   923→            onSubmit={form.handleSubmit(onSubmit)}
   924→            className="grid grid-cols-1 gap-3 sm:grid-cols-2"
   925→          >
   926→            <FormField
   927→              control={form.control}
   928→              name="riskLevel"
   929→              render={({ field }) => (
   930→                <FormItem>
   931→                  <FormLabel>Nivel de riesgo</FormLabel>
   932→                  <Select onValueChange={field.onChange} value={field.value}>
   933→                    <FormControl>
   934→                      <SelectTrigger className="w-full">
   935→                        <SelectValue />
   936→                      </SelectTrigger>
   937→                    </FormControl>
   938→                    <SelectContent>
   939→                      {RISK_LEVELS.map((r) => (
   940→                        <SelectItem key={r} value={r}>
   941→                          {riskLevelLabel(r)}
   942→                        </SelectItem>
   943→                      ))}
   944→                    </SelectContent>
   945→                  </Select>
   946→                  <FormMessage />
   947→                </FormItem>
   948→              )}
   949→            />
   950→            <FormField
