   128→   128→async function fetchFichas(
   129→   129→  filters: Record<string, string | undefined>,
   130→   130→): Promise<FichaFIBER[]> {
   131→   131→  const params = new URLSearchParams();
   132→   132→  for (const [k, v] of Object.entries(filters)) {
   133→   133→    if (v && v !== FILTER_ALL) params.set(k, v);
   134→   134→  }
   135→   135→  params.set('limit', '200');
   136→   136→  const res = await fetch(`/api/fichas?${params.toString()}`, {
   137→   137→    cache: 'no-store',
   138→   138→  });
   139→   139→  if (!res.ok) throw new Error('No se pudieron cargar las fichas');
   140→   140→  return (await res.json()) as FichaFIBER[];
   141→   141→}

   410→   410→function NewFichaDialog({
   411→   411→  open,
   412→   412→  onOpenChange,
   413→   413→  onCreated,
   414→   414→}: {
   415→   415→  open: boolean;
   416→   416→  onOpenChange: (v: boolean) => void;
   417→   417→  onCreated: () => void;
   418→   418→}) {
   419→   419→  const [submitting, setSubmitting] = useState(false);
   420→   420→  const form = useForm<NewFichaForm>({
   421→   421→    resolver: zodResolver(newFichaSchema),
   422→   422→    defaultValues: {
   423→   423→      fullName: '',
   424→   424→      rut: '',
   425→   425→      address: '',
   426→   426→      commune: 'Viña del Mar',
   427→   427→      lat: null,
   428→   428→      lng: null,
   429→   429→      phone: '',
   430→   430→      email: '',
   431→   431→      ageRange: null,
   432→   432→      riskLevel: 'bajo',
   433→   433→      vulnerability: '',
   434→   434→      householdSize: 1,
   435→   435→      hasPets: false,
   436→   436→      needsMedication: false,
   437→   437→      needsMobility: false,
   438→   438→      needsEvacuation: false,
   439→   439→      notes: '',
   440→   440→      assignedTo: '',
   441→   441→    },
   442→   442→  });
   443→   443→
   444→   444→  const onSubmit = async (values: NewFichaForm) => {
   445→   445→    setSubmitting(true);
   446→   446→    try {
   447→   447→      const body = {
   448→   448→        ...values,
   449→   449→        email: values.email === '' ? null : values.email,
   450→   450→        lat: values.lat === null || Number.isNaN(values.lat as number) ? null : values.lat,
   451→   451→        lng: values.lng === null || Number.isNaN(values.lng as number) ? null : values.lng,
   452→   452→      };
   453→   453→      const res = await fetch('/api/fichas', {
   454→   454→        method: 'POST',
   455→   455→        headers: { 'Content-Type': 'application/json' },
   456→   456→        body: JSON.stringify(body),
   457→   457→      });
   458→   458→      if (!res.ok) {
   459→   459→        const err = (await res.json().catch(() => null)) as {
   460→   460→          error?: string;
   461→   461→          issues?: { message: string }[];
   462→   462→        } | null;
   463→   463→        const msg =
   464→   464→          err?.error ??
   465→   465→          err?.issues?.[0]?.message ??
   466→   466→          'No se pudo crear la ficha.';
   467→   467→        toast.error('Error al crear ficha', { description: msg });
   468→   468→        return;
   469→   469→      }
   470→   470→      const created = (await res.json()) as FichaFIBER;
   471→   471→      toast.success('Ficha creada', {
   472→   472→        description: `Código ${created.code}`,
   473→   473→      });
   474→   474→      form.reset();
   475→   475→      onCreated();
   476→   476→    } catch {
   477→   477→      toast.error('Error de red al crear la ficha.');
   478→   478→    } finally {
   479→   479→      setSubmitting(false);
   480→   480→    }
   481→   481→  };
   482→   482→
