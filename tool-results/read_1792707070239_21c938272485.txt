   134→async function fetchFichas(
   135→  filters: Record<string, string | undefined>,
   136→): Promise<FichaFIBER[]> {
   137→  const params = new URLSearchParams();
   138→  for (const [k, v] of Object.entries(filters)) {
   139→    if (v && v !== FILTER_ALL) params.set(k, v);
   140→  }
   141→  params.set('limit', '200');
   142→  const res = await fetch(`/api/fichas?${params.toString()}`, {
   143→    cache: 'no-store',
   144→  });
   145→  if (!res.ok) throw new Error('No se pudieron cargar las fichas');
   146→  return (await res.json()) as FichaFIBER[];
   147→}

   413→// ---------------------------------------------------------------------------
   414→// New ficha dialog (react-hook-form + zod)
   415→// ---------------------------------------------------------------------------
   416→function NewFichaDialog({
   417→  open,
   418→  onOpenChange,
   419→  onCreated,
   420→}: {
   421→  open: boolean;
   422→  onOpenChange: (v: boolean) => void;
   423→  onCreated: () => void;
   424→}) {
   425→  const [submitting, setSubmitting] = useState(false);
   426→  const form = useForm<NewFichaForm>({
   427→    resolver: zodResolver(newFichaSchema),
   428→    defaultValues: {
   429→      fullName: '',
   430→      rut: '',
   431→      address: '',
   432→      commune: 'Viña del Mar',
   433→      lat: null,
   434→      lng: null,
   435→      phone: '',
   436→      email: '',
   437→      ageRange: null,
   438→      riskLevel: 'bajo',
   439→      vulnerability: '',
   440→      householdSize: 1,
   441→      hasPets: false,
   442→      needsMedication: false,
   443→      needsMobility: false,
   444→      needsEvacuation: false,
   445→      notes: '',
   446→      assignedTo: '',
   447→    },
   448→  });
   449→
   450→  const onSubmit = async (values: NewFichaForm) => {
   451→    setSubmitting(true);
   452→    try {
   453→      const body = {
   454→        ...values,
   455→        email: values.email === '' ? null : values.email,
   456→        lat: values.lat === null || Number.isNaN(values.lat as number) ? null : values.lat,
   457→        lng: values.lng === null || Number.isNaN(values.lng as number) ? null : values.lng,
   458→      };
   459→      const res = await fetch('/api/fichas', {
   460→        method: 'POST',
   461→        headers: { 'Content-Type': 'application/json' },
   462→        body: JSON.stringify(body),
   463→      });
   464→      if (!res.ok) {
   465→        const err = (await res.json().catch(() => null)) as {
   466→          error?: string;
   467→          issues?: { message: string }[];
   468→        } | null;
   469→        const msg =
   470→          err?.error ??
   471→          err?.issues?.[0]?.message ??
   472→          'No se pudo crear la ficha.';
   473→        toast.error('Error al crear ficha', { description: msg });
   474→        return;
   475→      }
   476→      const created = (await res.json()) as FichaFIBER;
   477→      toast.success('Ficha creada', {
   478→        description: `Código ${created.code}`,
   479→      });
   480→      form.reset();
   481→      onCreated();
   482→    } catch {
   483→      toast.error('Error de red al crear la ficha.');
   484→    } finally {
   485→      setSubmitting(false);
   486→    }
   487→  };
   488→
   489→  return (
   490→    <Dialog open={open} onOpenChange={onOpenChange}>
   491→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl">
   492→        <DialogHeader>
   493→          <DialogTitle>Nueva Ficha FIBER</DialogTitle>
   494→          <DialogDescription>
   495→            Registro individual de bienestar y riesgo. Datos sensibles bajo Ley
   496→            19.628 — requiere consentimiento informado del titular.
   497→          </DialogDescription>
   498→        </DialogHeader>
   499→
   500→        <Form {...form}>
   501→          <form
   502→            onSubmit={form.handleSubmit(onSubmit)}
   503→            className="grid grid-cols-1 gap-3 sm:grid-cols-2"
   504→          >
   505→            <FormField
   506→              control={form.control}
   507→              name="fullName"
   508→              render={({ field }) => (
   509→                <FormItem className="sm:col-span-2">
   510→                  <FormLabel>Nombre completo *</FormLabel>
   511→                  <FormControl>
   512→                    <Input placeholder="Ej. María González Pérez" {...field} />
   513→                  </FormControl>
   514→                  <FormMessage />
   515→                </FormItem>
   516→              )}
   517→            />
   518→            <FormField
   519→              control={form.control}
   520→              name="rut"
   521→              render={({ field }) => (
   522→                <FormItem>
   523→                  <FormLabel>RUT</FormLabel>
   524→                  <FormControl>
   525→                    <Input placeholder="12.345.678-9" {...field} />
   526→                  </FormControl>
   527→                  <FormMessage />
   528→                </FormItem>
   529→              )}
   530→            />
   531→            <FormField
   532→              control={form.control}
   533→              name="ageRange"
   534→              render={({ field }) => (
   535→                <FormItem>
   536→                  <FormLabel>Rango etario</FormLabel>
   537→                  <Select
   538→                    onValueChange={(v) => field.onChange(v as AgeRange)}
   539→                    value={field.value ?? undefined}
   540→                  >
   541→                    <FormControl>
   542→                      <SelectTrigger className="w-full">
   543→                        <SelectValue placeholder="Selecciona…" />
   544→                      </SelectTrigger>
   545→                    </FormControl>
   546→                    <SelectContent>
   547→                      {AGE_RANGES.map((r) => (
   548→                        <SelectItem key={r} value={r}>
   549→                          {r} años

   584→                      </SelectTrigger>
   585→                    </FormControl>
   586→                    <SelectContent>
   587→                      {COMMUNES.map((c) => (
   588→                        <SelectItem key={c} value={c}>
   589→                          {c}
   590→                        </SelectItem>
   591→                      ))}
   592→                    </SelectContent>
   593→                  </Select>
   594→                  <FormMessage />
   595→                </FormItem>
   596→              )}
   597→            />
   598→            <FormField
   599→              control={form.control}
   600→              name="riskLevel"
   601→              render={({ field }) => (
   602→                <FormItem>

   821→function EditFichaDialog({
   822→  ficha,
   823→  onClose,
   824→  onUpdated,
   825→}: {
   826→  ficha: FichaFIBER;
   827→  onClose: () => void;
   828→  onUpdated: () => void;
   829→}) {
   830→  const [submitting, setSubmitting] = useState(false);
   831→  const form = useForm<EditFichaForm>({
   832→    resolver: zodResolver(editFichaSchema),
   833→    defaultValues: {
   834→      riskLevel: ficha.riskLevel,
   835→      status: ficha.status,
   836→      assignedTo: ficha.assignedTo ?? '',
   837→      notes: ficha.notes ?? '',
   838→    },
   839→  });
   840→
   841→  // Reset form when ficha changes
   842→  useEffect(() => {
   843→    form.reset({
   844→      riskLevel: ficha.riskLevel,
   845→      status: ficha.status,
   846→      assignedTo: ficha.assignedTo ?? '',
   847→      notes: ficha.notes ?? '',
   848→    });
   849→  }, [ficha, form]);
   850→
   851→  const onSubmit = async (values: EditFichaForm) => {
   852→    setSubmitting(true);
   853→    try {
   854→      const body = {
   855→        riskLevel: values.riskLevel,
   856→        status: values.status,
   857→        assignedTo: values.assignedTo || null,
   858→        notes: values.notes || null,
   859→      };
   860→      const res = await fetch(`/api/fichas/${ficha.id}`, {
   861→        method: 'PATCH',
   862→        headers: { 'Content-Type': 'application/json' },
   863→        body: JSON.stringify(body),
   864→      });
   865→      if (!res.ok) {
   866→        const err = (await res.json().catch(() => null)) as {
   867→          error?: string;
   868→        } | null;
   869→        toast.error('Error al actualizar', {
   870→          description: err?.error ?? 'Intenta de nuevo.',
   871→        });
   872→        return;
   873→      }
   874→      toast.success('Ficha actualizada', {
   875→        description: `${ficha.code} · ${riskLevelLabel(values.riskLevel)} · ${fichaStatusLabel(
   876→          values.status,
   877→        )}`,
   878→      });
   879→      onUpdated();
   880→    } catch {
   881→      toast.error('Error de red al actualizar.');
   882→    } finally {
   883→      setSubmitting(false);
   884→    }
   885→  };
   886→
   887→  return (
   888→    <Dialog open onOpenChange={(v) => !v && onClose()}>
   889→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-xl">
   890→        <DialogHeader>
   891→          <DialogTitle className="flex items-center gap-2">
   892→            <FileText className="h-4 w-4 text-primary" />
   893→            {ficha.code}
   894→          </DialogTitle>
   895→          <DialogDescription>
   896→            {ficha.fullName} · {ficha.commune}
   897→          </DialogDescription>
   898→        </DialogHeader>
   899→
   900→        {/* Read-only detail */}
   901→        <div className="grid grid-cols-1 gap-2 rounded-md border bg-muted/30 p-3 text-xs sm:grid-cols-2">
   902→          {ficha.rut && (
   903→            <DetailRow icon={IdCard} label="RUT" value={ficha.rut} />
   904→          )}
   905→          {ficha.address && (
   906→            <DetailRow icon={MapPin} label="Dirección" value={`${ficha.address}, ${ficha.commune}`} />
   907→          )}
   908→          {ficha.phone && (
   909→            <DetailRow icon={Phone} label="Teléfono" value={ficha.phone} />
   910→          )}
   911→          {ficha.email && (
   912→            <DetailRow icon={Mail} label="Email" value={ficha.email} />
   913→          )}
   914→          {ficha.ageRange && (
   915→            <DetailRow icon={Users} label="Edad" value={`${ficha.ageRange} años`} />
   916→          )}

   918→            icon={Users}
   919→            label="Hogar"
   920→            value={`${ficha.householdSize} personas`}

   925→        </div>

   927→        <Form {...form}>
   928→          <form
   929→            onSubmit={form.handleSubmit(onSubmit)}
   930→            className="grid grid-cols-1 gap-3 sm:grid-cols-2"
   931→          >
   932→            <FormField
   933→              control={form.control}
   934→              name="riskLevel"
   935→              render={({ field }) => (
   936→                <FormItem>
   937→                  <FormLabel>Nivel de riesgo</FormLabel>
   938→                  <Select onValueChange={field.onChange} value={field.value}>
   939→                    <FormControl>
   940→                      <SelectTrigger className="w-full">
   941→                        <SelectValue />
   942→                      </SelectTrigger>

   945→                      {RISK_LEVELS.map((r) => (
   946→                        <SelectItem key={r} value={r}>
   947→                          {riskLevelLabel(r)}
   948→                        </SelectItem>
   949→                      ))}
   950→                    </SelectContent>
   951→                  </Select>
   952→                  <FormMessage />
