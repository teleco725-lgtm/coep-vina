    15→ * Server: socket.io on port 3003, path: '/' (REQUIRED for Caddy forwarding).
    16→ * Run with `bun --hot index.ts` for hot reload during development.

    86→function rand(min: number, max: number): number {
    87→  return Math.random() * (max - min) + min
    88→}
    89→function randInt(min: number, max: number): number {
    90→  return Math.floor(rand(min, max + 1))
    91→}
    92→function pick<T>(arr: readonly T[]): T {
    93→  return arr[Math.floor(Math.random() * arr.length)]
    94→}
    95→/**
    96→ * Pick a variant from the array filtered by critical-ness.
    97→ * `criticalOnly=true` returns only critical variants; `false` returns only
    98→ * non-critical ones. This is what keeps critical alerts globally sparse (~5%):
    99→ * the regular stream NEVER produces a critical variant; only the dedicated
   100→ * 5% path does.
   101→ */
   102→function pickVariant<T extends { severity: Severity }>(variants: T[], criticalOnly: boolean): T {
   103→  const filtered = variants.filter((v) => (v.severity === 'critical') === criticalOnly)
   104→  if (filtered.length === 0) {
   105→    // Fallback for sources without criticals (shouldn't normally happen).
   106→    return pick(variants)
   107→  }
   108→  return pick(filtered)
   109→}
   110→function round(v: number, d = 4): number {
   111→  const p = Math.pow(10, d)
   112→  return Math.round(v * p) / p
   113→}
   114→function randCoord(): { lat: number; lng: number } {
   115→  return {
   116→    lat: round(rand(VINA_LAT_RANGE.min, VINA_LAT_RANGE.max), 4),
   117→    lng: round(rand(VINA_LNG_RANGE.min, VINA_LNG_RANGE.max), 4),
   118→  }
   119→}
   120→function communeForCoord(): string {
   121→  const r = Math.random()
   122→  if (r < 0.55) return 'Viña del Mar'
   123→  if (r < 0.75) return 'Valparaíso'
   124→  if (r < 0.85) return 'Concón'
   125→  if (r < 0.93) return 'Quilpué'
   126→  return 'Villa Alemana'
   127→}
   128→function makeExternalId(prefix: string): string {
   129→  const year = new Date().getFullYear()
   130→  return `${prefix}-${year}-${randInt(1000, 9999)}`
   131→}

   133→// ============================================================================
   134→// History buffer (in-memory)
   135→// ============================================================================
   136→const HISTORY_MAX = 200
   137→const history: BroadcastAlert[] = []
   138→
   139→function pushAlert(alert: BroadcastAlert): void {
   140→  history.unshift(alert)
   141→  if (history.length > HISTORY_MAX) history.pop()
   142→}
   143→function getRecentAlerts(n: number): BroadcastAlert[] {
   144→  return history.slice(0, n)
   145→}

   654→// Regular (non-critical) generator pool.
   655→// SHOA marea alerts are produced by their own tide timer, so they are
   656→// NOT part of the random pool to avoid double-firing the same source.
   657→// Each generator is called WITHOUT `criticalOnly`, so it defaults to false
   658→// and only emits info/watch/warning variants (critical variants are filtered
   659→// out by `pickVariant`). This is what keeps critical alerts globally sparse.
   660→const REGULAR_GENERATORS: Array<() => BroadcastAlert> = [
   661→  genSenapred,
   662→  genConaf,
   663→  genShoaMet,
   664→  genGuardiaMarina,
   665→  genCbv2,
   666→  genCarabineros,
   667→  genRadio,
   668→]
   669→
   670→// Generators capable of producing critical variants.
   671→const CRITICAL_GENERATORS: Array<() => BroadcastAlert> = [
   672→  () => genSenapred(true),
   673→  () => genConaf(true),
   674→  () => genCbv2(true),
   675→  () => genCarabineros(true),
   676→]
   677→
   678→function generateAlert(): BroadcastAlert {
   679→  // Critical alerts must remain globally sparse (~5% of total).
   680→  if (Math.random() < 0.05) {
   681→    return generateCriticalAlert()
   682→  }
   683→  return pick(REGULAR_GENERATORS)()
   684→}
   685→
   686→function generateCriticalAlert(): BroadcastAlert {
   687→  // Pick a critical-capable source (SENAPRED, CONAF, CBV2 or Carabineros)
   688→  // and emit one of its critical variants (Alerta Roja, etc.).
   689→  return pick(CRITICAL_GENERATORS)()
   690→}

   693→// Stats computation
   694→// ============================================================================
   695→function computeStats(connectedClients: number): Stats {
   696→  const stats: Stats = {
   697→    total: history.length,
   698→    bySource: {},
   699→    bySeverity: {},
   700→    byCategory: {},
   701→    byState: {},
   702→    connectedClients,
   703→    generatedAt: new Date().toISOString(),
   704→  }
   705→  for (const a of history) {
   706→    stats.bySource[a.sourceCode] = (stats.bySource[a.sourceCode] ?? 0) + 1
   707→    stats.bySeverity[a.severity] = (stats.bySeverity[a.severity] ?? 0) + 1
   708→    stats.byCategory[a.category] = (stats.byCategory[a.category] ?? 0) + 1
   709→    stats.byState[a.state] = (stats.byState[a.state] ?? 0) + 1
   710→  }
   711→  return stats
   712→}

   824→// ============================================================================
   825→// Graceful shutdown (SIGTERM / SIGINT)
   826→// ============================================================================
   827→let shuttingDown = false
   828→function shutdown(signal: string): void {
   829→  if (shuttingDown) return
   830→  shuttingDown = true
   831→  console.log(`[alerts-service] received ${signal}, shutting down gracefully...`)
   832→
   833→  if (alertTimer) clearTimeout(alertTimer)
   834→  if (tideTimer) clearTimeout(tideTimer)
   835→  if (statsTimer) clearInterval(statsTimer)
   836→
   837→  // Close all open connections first, then the HTTP server.
   838→  io.close(() => {
   839→    httpServer.close(() => {
   840→      console.log('[alerts-service] server closed cleanly.')
   841→      process.exit(0)
   842→    })
   843→  })
   844→
   845→  // Failsafe: if for some reason close hangs, force exit after 5s.
   846→  setTimeout(() => {
   847→    console.warn('[alerts-service] force-exit after 5s timeout.')
   848→    process.exit(1)
   849→  }, 5000).unref()
   850→}
   851→
   852→process.on('SIGTERM', () => shutdown('SIGTERM'))
   853→process.on('SIGINT', () => shutdown('SIGINT'))
   854→process.on('uncaughtException', (err) => {
   855→  console.error('[alerts-service] uncaughtException:', err)
   856→})
   857→process.on('unhandledRejection', (reason) => {
   858→  console.error('[alerts-service] unhandledRejection:', reason)
   859→})
