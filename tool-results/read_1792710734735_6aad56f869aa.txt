    87→function rand(min: number, max: number): number {
    88→  return Math.random() * (max - min) + min
    89→}
    90→function randInt(min: number, max: number): number {
    91→  return Math.floor(rand(min, max + 1))
    92→}
    93→function pick<T>(arr: readonly T[]): T {
    94→  return arr[Math.floor(Math.random() * arr.length)]
    95→}
    96→/**
    97→ * Pick a variant from the array filtered by critical-ness.
    98→ * `criticalOnly=true` returns only critical variants; `false` returns only
    99→ * non-critical ones. This is what keeps critical alerts globally sparse (~5%):
   100→ * the regular stream NEVER produces a critical variant; only the dedicated
   101→ * 5% path does.
   102→ */
   103→function pickVariant<T extends { severity: Severity }>(variants: T[], criticalOnly: boolean): T {
   104→  const filtered = variants.filter((v) => (v.severity === 'critical') === criticalOnly)
   105→  if (filtered.length === 0) {
   106→    // Fallback for sources without criticals (shouldn't normally happen).
   107→    return pick(variants)
   108→  }
   109→  return pick(filtered)
   110→}
   111→function round(v: number, d = 4): number {
   112→  const p = Math.pow(10, d)
   113→  return Math.round(v * p) / p
   114→}

   119→  }
   120→}

   125→function makeExternalId(prefix: string): string {
   126→  const year = new Date().getFullYear()

   130→// ============================================================================
   131→// History buffer (in-memory)
   132→// ============================================================================
   133→const HISTORY_MAX = 200
   134→const history: BroadcastAlert[] = []
   135→
   136→function pushAlert(alert: BroadcastAlert): void {
   137→  history.unshift(alert)
   138→  if (history.length > HISTORY_MAX) history.pop()
   139→}
   140→function getRecentAlerts(n: number): BroadcastAlert[] {
   141→  return history.slice(0, n)
   142→}

   651→// Regular (non-critical) generator pool.
   652→// SHOA marea alerts are produced by their own tide timer, so they are
   653→// NOT part of the random pool to avoid double-firing the same source.
   654→// Each generator is called WITHOUT `criticalOnly`, so it defaults to false
   655→// and only emits info/watch/warning variants (critical variants are filtered
   656→// out by `pickVariant`). This is what keeps critical alerts globally sparse.
   657→const REGULAR_GENERATORS: Array<() => BroadcastAlert> = [
   658→  genSenapred,
   659→  genConaf,
   660→  genShoaMet,
   661→  genGuardiaMarina,
   662→  genCbv2,
   663→  genCarabineros,
   664→  genRadio,
   665→]
   666→
   667→// Generators capable of producing critical variants.
   668→const CRITICAL_GENERATORS: Array<() => BroadcastAlert> = [
   669→  () => genSenapred(true),
   670→  () => genConaf(true),
   671→  () => genCbv2(true),
   672→  () => genCarabineros(true),
   673→]
   674→
   675→function generateAlert(): BroadcastAlert {
   676→  // Critical alerts must remain globally sparse (~5% of total).
   677→  if (Math.random() < 0.05) {
   678→    return generateCriticalAlert()
   679→  }
   680→  return pick(REGULAR_GENERATORS)()
   681→}
   682→
   683→function generateCriticalAlert(): BroadcastAlert {
   684→  // Pick a critical-capable source (SENAPRED, CONAF, CBV2 or Carabineros)
   685→  // and emit one of its critical variants (Alerta Roja, etc.).
   686→  return pick(CRITICAL_GENERATORS)()
   687→}

   689→// ============================================================================
   690→// Stats computation
   691→// ============================================================================
   692→function computeStats(connectedClients: number): Stats {
   693→  const stats: Stats = {
   694→    total: history.length,
   695→    bySource: {},
   696→    bySeverity: {},
   697→    byCategory: {},
   698→    byState: {},
   699→    connectedClients,
   700→    generatedAt: new Date().toISOString(),
   701→  }
   702→  for (const a of history) {
   703→    stats.bySource[a.sourceCode] = (stats.bySource[a.sourceCode] ?? 0) + 1
   704→    stats.bySeverity[a.severity] = (stats.bySeverity[a.severity] ?? 0) + 1
   705→    stats.byCategory[a.category] = (stats.byCategory[a.category] ?? 0) + 1
   706→    stats.byState[a.state] = (stats.byState[a.state] ?? 0) + 1
   707→  }
   708→  return stats
   709→}

   821→// ============================================================================
   822→// Graceful shutdown (SIGTERM / SIGINT)
   823→// ============================================================================
   824→let shuttingDown = false
   825→function shutdown(signal: string): void {
   826→  if (shuttingDown) return
   827→  shuttingDown = true
   828→  console.log(`[alerts-service] received ${signal}, shutting down gracefully...`)
   829→
   830→  if (alertTimer) clearTimeout(alertTimer)
   831→  if (tideTimer) clearTimeout(tideTimer)
   832→  if (statsTimer) clearInterval(statsTimer)
   833→
   834→  // Close all open connections first, then the HTTP server.
   835→  io.close(() => {
   836→    httpServer.close(() => {
   837→      console.log('[alerts-service] server closed cleanly.')
   838→      process.exit(0)
   839→    })
   840→  })
   841→
   842→  // Failsafe: if for some reason close hangs, force exit after 5s.
   843→  setTimeout(() => {
   844→    console.warn('[alerts-service] force-exit after 5s timeout.')
   845→    process.exit(1)
   846→  }, 5000).unref()
   847→}
   848→
   849→process.on('SIGTERM', () => shutdown('SIGTERM'))
   850→process.on('SIGINT', () => shutdown('SIGINT'))
   851→process.on('uncaughtException', (err) => {
   852→  console.error('[alerts-service] uncaughtException:', err)
   853→})
   854→process.on('unhandledRejection', (reason) => {
   855→  console.error('[alerts-service] unhandledRejection:', reason)
   856→})
